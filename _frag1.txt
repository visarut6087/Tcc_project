227	def excel_to_text(excel_data):
228	
229	    output = []
230	
231	    for sheet in excel_data:
232	
233	        output.append(
234	            f"=== SHEET: {sheet['sheet']} ==="
235	        )
236	
237	        for row_number, row in enumerate(
238	            sheet["rows"],
239	            start=1
240	        ):
241	
242	            values = []
243	
244	            for col_number, value in enumerate(
245	                row,
246	                start=1
247	            ):

250	                    continue
251	
252	                values.append(
253	                    f"C{col_number}={value}"
254	                )
255	
256	            if values:
257	
258	                output.append(
259	                    f"ROW {row_number}: "
260	                    + " | ".join(values)
261	                )
262	
263	    return "\n".join(output)
